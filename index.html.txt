let tipo = 'Imóvel';
let valor = 200000;
let clientePorto = '';
let nomeAtual = '';
let whatsAtual = '';
let prazoAuto = 80;

// CONFIGURAÇÃO AUTOMÓVEL - PRAZO / TAXA / LIMITE CRÉDITO
const configAuto = {
  50:  { taxa: 0.20, min: 30000,  max: 50000,  padrao: 40000,  chips: [30000,40000,50000] },
  72:  { taxa: 0.20, min: 34000,  max: 65000,  padrao: 50000,  chips: [34000,50000,65000] },
  80:  { taxa: 0.18, min: 62500,  max: 125000, padrao: 80000,  chips: [62500,80000,100000,125000] },
  90:  { taxa: 0.17, min: 125000, max: 200000, padrao: 150000, chips: [125000,150000,200000] },
  120: { taxa: 0.16, min: 180000, max: 360000, padrao: 250000, chips: [180000,250000,360000] }
};

function taxaImovel(v){
  if(v <= 140000) return 0.27;
  if(v <= 280000) return 0.25;
  if(v <= 560000) return 0.23;
  return 0.215;
}
function taxaAuto(v){ return configAuto[prazoAuto].taxa; }
function getTaxa(){ return tipo === 'Imóvel' ? taxaImovel(valor) : taxaAuto(valor); }
function getPrazo(){ return tipo === 'Imóvel' ? 200 : prazoAuto; }
function getParcela(){ return Math.round((valor * (1 + getTaxa())) / getPrazo() * 100) / 100; }
function getTaxaMensal(){ return getTaxa() / getPrazo() * 100; }

function updateHero(){
  const hero = document.getElementById('hero');
  const title = document.getElementById('heroTitle');
  const sub = document.getElementById('heroSub');
  const pill = document.getElementById('heroPill');
  if(tipo === 'Imóvel'){
    hero.classList.remove('auto');
    title.innerHTML = 'Sua casa nova<br>sem pagar juros.';
    sub.innerText = 'Simule em 30 segundos e fale direto com Lucas Fortes.';
    pill.innerText = '🏠 Imóvel • 200 meses • Sem juros';
  } else {
    hero.classList.add('auto');
    title.innerHTML = 'Seu carro novo<br>sem pagar juros.';
    sub.innerText = 'Simule em 30 segundos e saia de carro novo.';
    pill.innerText = '🚗 Automóvel • ' + prazoAuto + ' meses • Sem juros';
  }
}

function selectPrazoAuto(prazo, el){
  prazoAuto = prazo;
  document.querySelectorAll('.prazo-opt').forEach(o=>o.classList.remove('selected'));
  if(el) el.classList.add('selected');
  const cfg = configAuto[prazo];
  const range = document.getElementById('valorRange');
  range.min = cfg.min; range.max = cfg.max; range.value = cfg.padrao; valor = cfg.padrao;
  document.getElementById('minLabel').innerText = 'R$ ' + cfg.min.toLocaleString('pt-BR');
  document.getElementById('maxLabel').innerText = 'R$ ' + cfg.max.toLocaleString('pt-BR');
  document.getElementById('tipoLabel').innerText = 'Ajuste a barra até o valor do seu automóvel - ' + prazo + ' meses.';
  renderChips(cfg.chips);
  updateValor(valor);
  updateHero();
}

function selectTipo(t, el){
  tipo = t;
  document.querySelectorAll('.opt').forEach(o=>o.classList.remove('selected'));
  el.classList.add('selected');
  updateHero();
  const range = document.getElementById('valorRange');
  const prazoBox = document.getElementById('prazoAutoBox');
  if(t==='Imóvel'){
    prazoBox.classList.remove('show');
    range.min=50000; range.max=800000; range.value=200000; valor=200000;
    document.getElementById('minLabel').innerText='R$ 50 mil';
    document.getElementById('maxLabel').innerText='R$ 800 mil';
    document.getElementById('tipoLabel').innerText='Ajuste a barra até o valor do seu imóvel.';
    renderChips([100000,140000,200000,350000,500000]);
  } else {
    prazoBox.classList.add('show');
    // Mantém o prazo já selecionado (padrão 80)
    selectPrazoAuto(prazoAuto, document.querySelector('.prazo-opt.selected'));
    document.getElementById('tipoLabel').innerText='Ajuste a barra até o valor do seu automóvel - ' + prazoAuto + ' meses.';
  }
  updateValor(valor);
}

function selectPorto(resposta, el){
  clientePorto = resposta;
  document.querySelectorAll('.porto-opt').forEach(o=>o.classList.remove('selected'));
  el.classList.add('selected');
  const promoBox = document.getElementById('promoBox');
  if(resposta === 'Sim'){
    promoBox.classList.add('show');
  } else {
    promoBox.classList.remove('show');
    document.getElementById('checkPromo').checked = false;
  }
}

function renderChips(list){
  document.getElementById('chips').innerHTML = list.map(v=>`<div class="chip" onclick="setValor(${v},this)">R$ ${v.toLocaleString('pt-BR')}</div>`).join('');
}
function setValor(v, el){
  document.querySelectorAll('.chip').forEach(c=>c.classList.remove('selected'));
  if(el) el.classList.add('selected');
  document.getElementById('valorRange').value=v;
  updateValor(v);
}
function updateValor(v){
  valor = parseInt(v);
  document.getElementById('valorTxt').innerText = 'R$ ' + valor.toLocaleString('pt-BR');
}
function next(n){
  if(n===3){
    let prazo = getPrazo();
    let parcela = getParcela();
    let taxaMensal = getTaxaMensal();
    document.getElementById('resumoTipo').innerText = `Consórcio de ${tipo} - R$ ${valor.toLocaleString('pt-BR')}`;
    document.getElementById('rCredito').innerText = 'R$ ' + valor.toLocaleString('pt-BR');
    document.getElementById('rParcela').innerText = 'R$ ' + parcela.toLocaleString('pt-BR',{minimumFractionDigits:2}) + ' / mês';
    document.getElementById('rPrazo').innerText = prazo + ' meses';
    // AUTOMÓVEL NÃO EXIBE TAXA
    const taxaRow = document.getElementById('rTaxaRow');
    if(tipo === 'Automóvel'){
      taxaRow.style.display = 'none';
    } else {
      taxaRow.style.display = 'flex';
      document.getElementById('rTaxaMensal').innerText = taxaMensal.toLocaleString('pt-BR',{minimumFractionDigits:3, maximumFractionDigits:3}) + '% a.m.';
    }
  }
  document.querySelectorAll('.step').forEach(s=>s.classList.remove('active'));
  document.getElementById('step'+n).classList.add('active');
  window.scrollTo({top:0,behavior:'smooth'});
}
function restart(){
  document.getElementById('nome').value = '';
  document.getElementById('whats').value = '';
  clientePorto = '';
  document.querySelectorAll('.porto-opt').forEach(o=>o.classList.remove('selected'));
  document.getElementById('promoBox').classList.remove('show');
  document.getElementById('checkPromo').checked = false;
  tipo = 'Imóvel';
  prazoAuto = 80;
  document.querySelectorAll('.opt').forEach(o=>o.classList.remove('selected'));
  document.querySelector('.opt').classList.add('selected');
  selectTipo('Imóvel', document.querySelector('.opt'));
  next(1);
}
